% Blocco bottleneck di ResNet-50.
% Numeri: He et al. 2016, Fig. 5 destra (256-d, 1x1 64, 3x3 64, 1x1 256, somma con lo shortcut).
% TorchVision v0.21.0, resnet.py, Bottleneck.forward: conv1, conv2, conv3, out += identity.
% Batch normalization e ReLU non sono disegnate, lo dice la didascalia.
\begin{tikzpicture}[
    font=\footnotesize,
    node distance=0.55cm,
    port/.style={draw=black!60, rounded corners=1pt, fill=white, align=center, minimum height=1.5cm, minimum width=1.45cm, inner sep=2pt},
    wide/.style={draw=black!60, rounded corners=1pt, fill=orange!25, align=center, minimum height=1.5cm, minimum width=1.7cm, inner sep=2pt},
    narrow/.style={wide, minimum height=0.9cm},
    sum/.style={draw=black!70, circle, fill=white, inner sep=1pt, minimum size=0.45cm},
    arr/.style={-{Stealth[length=4pt]}, black!70},
  ]
  \node[port] (in) {input\\ 256 ch.};
  \node[wide, right=of in] (c1) {$1\times1$ conv.\\ 64 ch.};
  \node[narrow, right=of c1] (c2) {$3\times3$ conv.\\ 64 ch.};
  \node[wide, right=of c2] (c3) {$1\times1$ conv.\\ 256 ch.};
  \node[sum, right=of c3] (s) {$+$};
  \node[port, right=of s] (out) {output\\ 256 ch.};
  \foreach \a/\b in {in/c1, c1/c2, c2/c3, c3/s, s/out} \draw[arr] (\a) -- (\b);
  \coordinate (split) at ($(in.east)!0.5!(c1.west)$);
  \fill[black!70] (split) circle (1.2pt);
  \draw[arr, rounded corners=3pt] (split) -- ++(0,1.2) -| (s.north);
  \node[above, black!80] at ($(split)!0.5!(s.center |- split)+(0,1.2)$) {shortcut connection: the input is added to the output of the convolutions};
\end{tikzpicture}%
