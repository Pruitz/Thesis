% Architettura di ResNet-50 come backbone di fcn_resnet50 e deeplabv3_resnet50.
% Numeri: He et al. 2016, Table 1 (colonna 50-layer) e TorchVision v0.21.0
% (resnet.py: Bottleneck, [3, 4, 6, 3], expansion 4, _make_layer: dilation *= stride;
% fcn.py/deeplabv3.py: replace_stride_with_dilation=[False, True, True], return_layers fino a layer4).
% Dimensioni per un ingresso 224x224: conv1 112, maxpool 56, layer1 56, layer2 28, layer3 28, layer4 28.
\begin{tikzpicture}[
    font=\scriptsize,
    node distance=0.22cm,
    blk/.style={draw=black!60, rounded corners=1pt, align=center, minimum height=1.45cm, minimum width=1.55cm, inner sep=2pt},
    inp/.style={blk, fill=white},
    stem/.style={blk, fill=black!8},
    res/.style={blk, fill=orange!25},
    atr/.style={blk, fill=orange!55},
    cut/.style={blk, draw=black!40, dashed, text=black!50},
    arr/.style={-{Stealth[length=3pt]}, black!70},
    os/.style={font=\scriptsize, black!75, align=center},
  ]
  \node[inp]  (in) {input\\ RGB image\\ 3 ch.};
  \node[stem, right=of in] (c1) {conv1\\ kernel $7\times7$\\ stride 2\\ 64 ch.};
  \node[stem, right=of c1] (mp) {max pool\\ $3\times3$\\ stride 2\\ 64 ch.};
  \node[res,  right=of mp] (l1) {\texttt{layer1}\\ 3 blocks\\ 256 ch.};
  \node[res,  right=of l1] (l2) {\texttt{layer2}\\ 4 blocks\\ 512 ch.};
  \node[atr,  right=of l2] (l3) {\texttt{layer3}\\ 6 blocks\\ 1024 ch.\\ rate 2};
  \node[atr,  right=of l3] (l4) {\texttt{layer4}\\ 3 blocks\\ 2048 ch.\\ rate 4};
  \node[cut,  right=of l4] (fc) {avg pool\\ fc 1000};
  \foreach \a/\b in {in/c1, c1/mp, mp/l1, l1/l2, l2/l3, l3/l4} \draw[arr] (\a) -- (\b);
  \draw[arr, dashed, black!40] (l4) -- (fc);
  \node[os, anchor=east] at ($(in.south west)+(-0.1,-0.35)$) {OS};
  \node[os, anchor=east] at ($(in.south west)+(-0.1,-0.8)$) {input size};
  % spazio vuoto a destra, largo quanto l'etichetta a sinistra, per centrare la fila delle caselle
  \node[os, anchor=west] at ($(fc.south east)+(0.1,-0.8)$) {\phantom{input size}};
  \foreach \n/\t/\s in {in/1/224, c1/2/112, mp/4/56, l1/4/56, l2/8/28, l3/\textbf{8}/\textbf{28}, l4/\textbf{8}/\textbf{28}} {
    \node[os] at ($(\n.south)+(0,-0.35)$) {\t};
    \node[os] at ($(\n.south)+(0,-0.8)$) {$\s\times\s$};
  }
  \node[os, text=black!50] at ($(fc.south)+(0,-0.55)$) {removed};
\end{tikzpicture}%
